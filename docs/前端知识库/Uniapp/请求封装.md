# uniapp 请求封装

## 1. 简易版本

### 1.1 utils/request.js

```js title="utils/request.js"
// utils/request.js

const BASE_URL = import.meta.env.VITE_APP_BASE_URL
import { TokenKey } from '@/constant/index.ts'

// 刷新状态管理
let isRefreshing = false
let requestQueue = [] // 存储等待的请求

// 封装请求失败处理
const requestFailHandler = err => {
    let errMsg = '网络异常，请稍后重试'
    if (err.errMsg?.includes('timeout')) {
        errMsg = '请求超时，请检查网络'
    } else if (err.errMsg?.includes('cross-domain')) {
        errMsg = '跨域错误，请配置跨域白名单'
    } else if (err.statusCode === 404) {
        errMsg = '接口地址不存在'
    } else if (err.statusCode === 500) {
        errMsg = '服务端内部错误'
    }
    uni.showToast({ title: errMsg, icon: 'none' })
    return Promise.reject(err)
}

// 刷新 token
const refreshTokenFn = async () => {
  if (isRefreshing) {
    // 已经在刷新，返回同一个Promise
    return refreshPromise
  }
  try {
    isRefreshing = true
    const { useUserStore } = await import('@/stores/user')
    const userStore = useUserStore()
    console.log('userStore', userStore)
    refreshPromise = userStore.reFreshTokenHandler()
    const result = await refreshPromise
    return result
  } finally {
    isRefreshing = false
    refreshPromise = null
  }
}

// 拦截器配置
const interceptors = {
    // 请求拦截
    request(config) {
        const token = uni.getStorageSync(TokenKey)
        if (token) {
            config.header = {
                ...config.header,
                Authorization: `Bearer ${token}`
            }
        }
        return config
    },

    // 响应拦截
    async response(response, config) {
        const { data, statusCode } = response

        if (statusCode >= 200 && statusCode < 300) {
            // token 过期
            if (data?.code === 401) {
                // 刷新自身失败，强制登录
                if (config.url.endsWith('/auth/refresh')) {
                  uni.showToast({ title: '登录过期，请重新登录', icon: 'none' })
                  uni.removeStorageSync(TokenKey)
                  uni.redirectTo({
                    url: '/pages/login/login'
                  })
                  return Promise.reject('refresh token failed')
                }

                // 不在刷新 → 开始刷新
                if (!isRefreshing) {
                  refreshTokenFn()
                    .then(() => {
                      // 刷新成功 → 重试所有队列中的请求
                      requestQueue.forEach(cb => cb())
                      requestQueue = []
                    })
                    .catch(() => {
                      // 刷新失败 → 清空队列并跳登录
                      const failedQueue = [...requestQueue]
                      requestQueue = []

                      // 所有等待的请求都失败
                      failedQueue.forEach(({ reject }) => {
                        reject(new Error('Token刷新失败,请重新登录'))
                      })

                      uni.showToast({ title: '登录已过期，请重新登录', icon: 'none' })
                      uni.removeStorageSync(TokenKey)
                      uni.redirectTo({ url: '/pages/login/login' })
                    })
                }

                // 进入等待队列
                return new Promise((resolve, reject) => {
                  requestQueue.push({
                    resolve: () => resolve(retryRequestFn(config)),
                    reject
                  })
                })

                uni.showToast({ title: '登录过期，请重新登录', icon: 'none' })
                uni.removeStorageSync(TokenKey)
                setTimeout(() => {
                    uni.redirectTo({
                        url: '/pages/login/login'
                    })
                }, 500)
                return Promise.reject('登录过期，请重新登录')
            }

            return data
        }

        // 非 200 响应
        uni.showToast({ title: data?.msg || '请求错误', icon: 'none' })
        return Promise.reject(data)
    },

    // 错误拦截
    error(error) {
        requestFailHandler(error)
    }
}

// 请求重试
function retryRequestFn(config) {
    return new Promise((resolve, reject) => {
        const token = uni.getStorageSync(TokenKey)
        uni.request({
            ...config,
            header: {
                ...config.header,
                Authorization: `Bearer ${token}`
            },
            success: res => {
                try {
                    // 直接处理响应，避免再次进入拦截器循环
                    if (res.statusCode >= 200 && res.statusCode < 300) {
                        resolve(res.data)
                    } else {
                        // 如果还是401，说明刷新后的token还是无效，直接跳登录
                        if (res.data?.code === 401) {
                            uni.showToast({ title: '登录已过期，请重新登录', icon: 'none' })
                            uni.removeStorageSync(TokenKey)
                            uni.redirectTo({ url: '/pages/login/login' })
                            reject(new Error('Token无效,请重新登录'))
                        } else {
                            uni.showToast({ title: res.data?.msg || '请求错误', icon: 'none' })
                            reject(res.data)
                        }
                    }
                } catch (err) {
                    reject(err)
                }
            },
            fail: err => {
                requestFailHandler(err)
                reject(err)
            }
        })
    })
}

// 通用请求封装
function request(options) {
    const config = interceptors.request({
        url: BASE_URL + options.url,
        method: options.method || 'GET',
        data: options.data || {},
        header: options.header || {},
        timeout: options.timeout || 30000
    })

    return new Promise((resolve, reject) => {
        uni.request({
            ...config,
            success: res => {
                try {
                    const data = interceptors.response(res, config)
                    resolve(data)
                } catch (err) {
                    reject(err)
                }
            },
            fail: err => {
                interceptors.error(err)
                reject(err)
            }
        })
    })
}

// 通用文件上传封装
function upload(options) {
    const config = interceptors.request({
        url: BASE_URL + options.url,
        filePath: options.filePath,
        name: options.name || 'file',
        formData: options.formData || {},
        header: options.header || {},
        timeout: options.timeout || 30000
    })

    return new Promise((resolve, reject) => {
        uni.uploadFile({
            ...config,
            success: res => {
                try {
                    // 解析响应数据
                    const data = res.data ? JSON.parse(res.data) : {}
                    // 应用响应拦截器
                    const result = interceptors.response({ ...res, data }, config)
                    resolve(result)
                } catch (err) {
                    reject(err)
                }
            },
            fail: err => {
                interceptors.error(err)
                reject(err)
            }
        })
    })
}

// 常用 CRUD 封装
const http = {
    get(url, params = {}, config = {}) {
        return request({
            url,
            method: 'GET',
            data: params,
            ...config
        })
    },

    post(url, data = {}, config = {}) {
        return request({
            url,
            method: 'POST',
            data,
            ...config
        })
    },

    put(url, data = {}, config = {}) {
        return request({
            url,
            method: 'PUT',
            data,
            ...config
        })
    },

    delete(url, data = {}, config = {}) {
        return request({
            url,
            method: 'DELETE',
            data,
            ...config
        })
    },

    // 文件上传方法
    upload(url, filePath, name = 'file', formData = {}, config = {}) {
        return upload({
            url,
            filePath,
            name,
            formData,
            ...config
        })
    }
}

export default http


```

## 2. 使用示例

### 2.1 请求封装

```js  title='api/user.js'
import http from '@/utils/request.js';

export function getUserList(params) {
    return http.get('/user/list', params);
}

```

### 2.2 在页面中使用
```js
// 在任意页面或模块中使用
import http from '@/utils/request.js';

async function loadData() {
  try {
    const res = await http.get('/user/list', { page: 1 });
    console.log('用户列表：', res);
  } catch (e) {
    console.error('请求失败：', e);
  }
}

```
